import React, { useState, useEffect, useCallback } from 'react';
// 引入antd组件库
import {
  Button,
  Radio,
  Card,
  Icon,
  Input,
  Divider,
  Popconfirm,
  DatePicker,
  Table,
} from 'antd';
import moment from 'moment';
import { useHistory } from 'react-router-dom';
import Breadcrumb from 'components/breadcrumb';
import { ROUTEBASE } from 'common/constant';
import { getQuery } from 'common/utils';
import {
  getEmailsList,
  deleteEmail,
  revokeEmail,
  interruptEmail,
} from 'apis/emails';
import { Status, StatusTranslation } from './enums';

const PAGE_SIZE = 10;

export default function EmailsHome() {
  const initStartTime = moment(+getQuery('startTime')).valueOf() ? +getQuery('startTime') : null;
  const initEndTime = moment(+getQuery('endTime')).valueOf() ? +getQuery('endTime') : null;
  const initStatus = getQuery('status') ? +getQuery('status') : Status.DONE;
  const initSubject = getQuery('subject') || '';
  const initPage = getQuery('page') ? +getQuery('page') : 1;

  const [status, setStatus] = useState(initStatus);
  const [subject, setSubject] = useState(initSubject);
  const [sendDates, setSendDates] = useState([initStartTime, initEndTime]);
  const [page, setPage] = useState(initPage);
  const [total, setTotal] = useState(0);
  const [emailsList, setEmailsList] = useState([]);
  const history = useHistory();

  const fetchList = useCallback(() => {
    (async () => {
      const { data: { developerEmails, total } } = await getEmailsList({
        subject,
        startTime: sendDates[0],
        endTime: sendDates[1],
        pageNum: page - 1,
        pageSize: PAGE_SIZE,
        status,
      });
      setTotal(total);
      setEmailsList(developerEmails);
    })();
  }, [status, page, sendDates, subject]);

  const handleClickAddBtn = () => {
    history.push(`${ROUTEBASE}/emails/edit`);
  };

  const handleClickEditBtn = (id) => {
    history.push(`${ROUTEBASE}/emails/edit?id=${id}`);
  };

  const handleClickDeleteBtn = async (id) => {
    await deleteEmail({ id });
    fetchList();
  };

  const handleClickInterruptBtn = async (id) => {
    await interruptEmail({ id });
    fetchList();
  };

  const handleClickRevokeBtn = async (id) => {
    await revokeEmail({ id });
    fetchList();
  };

  const handleClickViewBtn = (id) => {
    history.push(`${ROUTEBASE}/emails/preview?id=${id}`);
  };

  useEffect(() => {
    fetchList();
  }, [fetchList]);

  const onPageChange = (page) => {
    setPage(page);
  };

  useEffect(() => {
    const startTime = sendDates[0] || '';
    const endTime = sendDates[1] || '';
    let timequery = '';
    if (startTime) timequery += `&startTime=${startTime}`;
    if (endTime) timequery += `&endTime=${endTime}`;
    history.push(`${ROUTEBASE}/emails?status=${status}&page=${page}&subject=${subject}${timequery}`);
  }, [history, page, sendDates, subject, status]);

  return (
    <div className="base-container">
      <Breadcrumb data={['开发者服务', '开发者邮件']} />
      <Card>
        <div style={{ marginBottom: 20 }}>
          <Radio.Group
            value={status}
            onChange={(e) => {
              setStatus(e.target.value);
            }}
          >
            {
              Object.keys(Status).map((key) => <Radio.Button value={Status[key]} key={key}>{StatusTranslation[key]}</Radio.Button>)
            }
          </Radio.Group>
          <DatePicker.RangePicker
            style={{ marginLeft: 20 }}
            showTime
            defaultValue={[
              moment(sendDates[0]).valueOf() ? moment(sendDates[0]) : null,
              moment(sendDates[1]).valueOf() ? moment(sendDates[1]) : null,
            ]}
            onChange={(dates) => {
              const [startTime, endTime] = dates;
              setSendDates([startTime && startTime.valueOf(), endTime && endTime.valueOf()]);
            }}
          />
          <Input.Search
            placeholder="按邮件标题搜索"
            enterButton="搜索"
            style={{ width: 200, marginRight: 20, marginLeft: 20 }}
            onSearch={(value) => {
              setSubject(value);
            }}
          />
          <Button type="primary" style={{ float: 'right' }} onClick={handleClickAddBtn}>
            <Icon type="plus" />
            新增邮件
          </Button>
        </div>
        <Table
          dataSource={emailsList}
          rowKey="id"
          pagination={{
            current: page,
            size: 'small',
            pageSize: PAGE_SIZE,
            onChange: onPageChange,
            total,
          }}
        >
          <Table.Column
            title="发送时间"
            dataIndex="sendTime"
            render={(sendTime) => moment(sendTime).format('YYYY-MM-DD HH:mm')}
          />
          <Table.Column title="操作人" dataIndex="operator" />
          <Table.Column title="邮件标题" dataIndex="subject" />
          <Table.Column
            title="发送对象"
            dataIndex="emailType"
            render={(type) => ['', '全部国内开发者', '全部海外开发者', '指定开发者邮件', '通配邮件'][type]}
          />
          <Table.Column title="发送量" dataIndex="totalQuantity" />
          <Table.Column title="成功发送量" dataIndex="successQuantity" />
          <Table.Column title="未成功发送量" dataIndex="failedQuantity" />
          <Table.Column
            title="操作"
            render={(_, { id, status }) => {
              let operateCell;
              switch (status) {
                case Status.DRAFT:
                  operateCell = (
                    <>
                      <Button type="link" style={{ padding: 0 }} onClick={() => handleClickEditBtn(id)}>编辑</Button>
                      <Divider type="vertical" />
                      <Popconfirm
                        title="确认要删除吗？"
                        onConfirm={() => { handleClickDeleteBtn(id); }}
                        onCancel={() => { }}
                        okText="是的"
                        cancelText="我手滑了"
                      >
                        <Button type="link" style={{ padding: 0 }}>删除</Button>
                      </Popconfirm>
                    </>
                  );
                  break;
                case Status.SENDING:
                  operateCell = (
                    <>
                      <Button onClick={() => handleClickInterruptBtn(id)}>停止</Button>
                    </>
                  );
                  break;
                case Status.WAIT_SEND:
                  operateCell = (
                    <>
                      <Button type="link" style={{ padding: 0 }} onClick={() => handleClickEditBtn(id)}>编辑</Button>
                      <Divider type="vertical" />
                      <Popconfirm
                        title="确认要撤销吗？"
                        onConfirm={() => { handleClickRevokeBtn(id); }}
                        onCancel={() => { }}
                        okText="是的"
                        cancelText="我手滑了"
                      >
                        <Button type="link" style={{ padding: 0 }}>撤销</Button>
                      </Popconfirm>
                    </>
                  );
                  break;
                case Status.DONE:
                  operateCell = (
                    <>
                      <Button style={{ padding: 0 }} type="link" onClick={() => handleClickViewBtn(id)}>查看</Button>
                    </>
                  );
                  break;
                default:
                  break;
              }
              return operateCell;
            }}
          />
        </Table>
      </Card>
    </div>
  );
}
