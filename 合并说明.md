## 说明
1. 在views文件夹加入student文件夹
2. 在components文件夹加入echart文件夹
3. 在config文件夹的router.config.js加入
```js
      // 学业预测
      {
        path: '',
        component: RouteView,
        redirect: '/student/studentone',
        name: 'studentone',
        meta: { title: '学业评估', icon: 'user', keepAlive: true, permission: ['user'] },
        children: [
          {
            path: '/student/studentone',
            name: 'studentone',
            component: () => import('@/views/student/Studentone'),
            meta: { title: '学业一', keepAlive: true, permission: ['user'] }
          },
          {
            path: '/student/studenttwo',
            name: 'studenttwo',
            component: () => import('@/views/student/Studenttwo'),
            meta: { title: '学业二', keepAlive: true, permission: ['user'] }
          }
        ]
      }
```
