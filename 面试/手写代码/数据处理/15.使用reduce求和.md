function flat(arr, depth) {
		while(arr.some(item => Array.isArray(item)) && depth > 0) {
		arr = [].concat(...arr)
		depth--
	}
	return arr
}

function flat(arr) {
	let str = JSON.stringify(arr)
	str = str.replace(/(\[|\])/g, '')
	str = `[${str}]`
	return JSON.parse(str)
}
