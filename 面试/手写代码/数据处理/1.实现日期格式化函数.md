## ES6

```js
var dateFormat = function(dataInput, format) {
	const day = String(dataInput.getDate()).padStart(2, '0')
	const month = String(dataInput.getMonth() + 1).padStart(2, '0')
    console.log(typeof month)
	const year = dataInput.getFullYear()
	format = format.replace(/yyyy/, year).replace(/MM/, month).replace(/dd/, day)
	return format
}

console.log(dateFormat(new Date('2020-12-01'),'yyyy/MM/dd'))// 2020/12/01
console.log(dateFormat(new Date('2020-04-01'),'yyyy/MM/dd')) // 202/04/01
dateFormat(new Date('2020-04-01'),'yyyy年MM月dd日') // 2020年04月01日
```


## ES5
```js
var dateFormat = function(dataInput, format) {
	const day = (dataInput.getDate()) < 10 ?  '0' + dataInput.getDate() : dataInput.getDate()
	const month = (dataInput.getMonth() + 1) < 10 ? '0' + (dataInput.getMonth() + 1) : (dataInput.getMonth() + 1)
    console.log(typeof month)
	const year = dataInput.getFullYear()
	format = format.replace(/yyyy/, year).replace(/MM/, month).replace(/dd/, day)
	return format
}

console.log(dateFormat(new Date('2020-12-01'),'yyyy/MM/dd'))// 2020/12/01
console.log(dateFormat(new Date('2020-04-01'),'yyyy/MM/dd')) // 202/04/01
dateFormat(new Date('2020-04-01'),'yyyy年MM月dd日') // 2020年04月01日

```
