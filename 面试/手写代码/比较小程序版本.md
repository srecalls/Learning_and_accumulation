      function compare(v1,v2) {
       v1 = v1.split('.')
       v2 = v2.split('.')
       const max = Math.max(v1.length,v2.length)

       while(v1.length<max) {
         v1.push('0')
       }

       while(v2.length<max) {
         v2.push('0')
       }

  
       for(let i = 0;i<max;i++){
        let num1 = parseInt(v1[i])
        let num2 = parseInt(v2[i])
  
        if(num1 > num2) {
          return 1
        } else if (num1 <num2) {
          return -1
        }
       }
       return 0
      }
