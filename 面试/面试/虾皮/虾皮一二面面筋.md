3.实现函数a，奇数次调用打印1，偶数次调用打印2，不能使用额外的全局变量。  
卡了，没想出来，面试官提醒用闭包，然后用闭包写出来了。然后面试官说不能有别的全局变量，问我知不知道立即执行函数，最终写出来了，总之就是全靠面试官提示。  
  [[1.对闭包的理解]]

要实现一个函数 `a`，使得奇数次调用时打印 `1`，偶数次调用时打印 `2`，并且不使用额外的全局变量，可以利用函数的闭包性质来实现。闭包允许函数在其外部访问和更新内部状态。

下面是一种实现方式：

```javascript
function createCounter() {
  let count = 0; // 内部状态，用于记录调用次数

  function a() {
    count++; // 每次调用增加计数器

    if (count % 2 === 1) {
      console.log(1); // 奇数次调用打印 1
    } else {
      console.log(2); // 偶数次调用打印 2
    }
  }

  return a; // 返回函数 a
}

const a = createCounter(); // 创建计数器函数

a(); // 输出 1
a(); // 输出 2
a(); // 输出 1
a(); // 输出 2
// 以此类推，每次调用的结果会依次是 1、2、1、2...
```

在上述实现中，`createCounter` 函数内部定义了一个局部变量 `count`，用于记录调用次数。然后，它返回一个匿名函数 `a`，该函数可以访问和更新 `count` 变量。

每次调用 `a` 函数时，都会增加 `count` 的值。根据 `count` 的值是奇数还是偶数，函数 `a` 打印相应的结果。

通过这种方式，我们利用了闭包的特性，避免了使用额外的全局变量来记录调用次数，实现了奇偶交替打印的效果。

const allRequests = [
  "http://jsonplaceholder.typicode.com/posts/1",
  "http://jsonplaceholder.typicode.com/posts/2",
  "http://jsonplaceholder.typicode.com/posts/3",
  "http://jsonplaceholder.typicode.com/posts/4",

];

function sendRequest(urls,max,callback) {
  const urlsLen = urls.length;
  const promiseAll = [];
  // 最后所有的结果
  const res = new Array(urlsLen).fill(0);

  let curIndex = -1;

  function fetchSingleUrl(curRes) {
    // 这里获取到的curIndex其实是错误的，无法确定当前返回的结果是该url获取的
    console.log(999,curIndex)
    res[curIndex] = curRes;
    curIndex++;
    if(curIndex >= urlsLen) {
      callback(res);
    }else {
      fetch(urls[curIndex]).then(fetchSingleUrl).catch(fetchSingleUrl);
    }
  }
  // 如果urls总共比max还小，那么就使用Promise.all一起发出去
  if(urlsLen <= max) {
    for(let i =0;i<urlsLen;i++) {
      promiseAll.push(fetch(urls[i]));
    }
    Promise.all(promiseAll).then(curRes => {
      console.log(112,curRes)
      callback(curRes);
    }).catch((err)=> {
      console.log(333,err)
      callback(err);
    })
  }else {
    for(let i =0;i<max;i++) {
      fetch(urls[i]).then(fetchSingleUrl).catch(fetchSingleUrl)
    }
  }
}

sendRequest(allRequests, 2, (result) => console.log(result));
