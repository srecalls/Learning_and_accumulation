手写一个前端计时器UI  
```js
    <div>
      <span id="hour">00</span>: <span id="minute">00</span>:
      <span id="second">00</span>
    </div>
    <div>
      <button onclick="start()">start</button>
      <button onclick="stop()">end</button>
      <button onclick="reset()">reset</button>
    </div>
    <script>
      let hour = 0;
      let minute = 0;
      let second = 0;
      let timeInterval = null;
      function start() {
        timeInterval = setInterval(updateTime, 1000);
      }

      function stop() {
        clearInterval(timeInterval);
      }
      function reset() {
        clearInterval(timeInterval);
        hour = 0;
        minute = 0;
        second = 0;
        showUI();
      }
      function updateTime() {
        second++;
        if (second === 60) {
          minute++;
          second = 0;
        }
        if (minute === 60) {
          hour++;
          second = 0;
        }
        if (hour === 24) {
          hour = 0;
        }
        showUI();
      }
      function showUI() {
        let hourText = document.getElementById('hour');
        let minuteText = document.getElementById('minute');
        let secondText = document.getElementById('second');
        hourText.textContent = hour.toString().padStart('2', 0);
        minuteText.textContent = minute.toString().padStart('2', 0);
        secondText.textContent = second.toString().padStart('2', 0);
      }
    </script>
```
