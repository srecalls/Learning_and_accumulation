下面是一个手写的场景题，描述了对一组图片进行加载，并在超时时抛出错误的情况：

```javascript
// 加载一组图片，并在超时时抛出错误
function loadImages(imageUrls, timeout) {
  return new Promise((resolve, reject) => {
    const images = [];
    let loadedCount = 0;
    let isTimeout = false;

    // 超时处理
    const timeoutId = setTimeout(() => {
      isTimeout = true;
      reject(new Error('Image loading timed out'));
    }, timeout);

    // 图片加载完成回调函数
    function imageLoadedHandler() {
      loadedCount++;

      if (loadedCount === imageUrls.length && !isTimeout) {
        clearTimeout(timeoutId);
        resolve(images);
      }
    }

    // 逐个加载图片
    for (let i = 0; i < imageUrls.length; i++) {
      const img = new Image();
      img.onload = imageLoadedHandler;
      img.onerror = imageLoadedHandler;
      img.src = imageUrls[i];
      images.push(img);
    }
  });
}

// 示例用法
const urls = [
  'image1.jpg',
  'image2.jpg',
  'image3.jpg'
];
const timeout = 5000; // 5秒超时

loadImages(urls, timeout)
  .then(images => {
    console.log('All images loaded:', images);
  })
  .catch(error => {
    console.error('Error loading images:', error);
  });
```

在上述示例中，`loadImages` 函数接收一组图片的 URL 数组 `imageUrls` 和超时时间 `timeout` 作为参数。它返回一个 Promise 对象，用于处理图片加载完成或超时时的结果。

在函数内部，创建了一个空数组 `images` 用于存储加载完成的图片对象。`loadedCount` 记录已加载完成的图片数量，`isTimeout` 标志表示是否超时。

通过循环遍历 `imageUrls` 数组，逐个创建 `Image` 对象，并为其设置 `onload` 和 `onerror` 事件处理函数。每当图片加载完成（无论是成功加载还是加载出错），都会调用 `imageLoadedHandler` 函数进行处理。

在 `imageLoadedHandler` 函数中，已加载的图片数量 `loadedCount` 会自增，然后检查是否所有图片都已加载完成且未超时。如果是，则清除超时计时器，调用 `resolve` 函数将加载完成的图片数组 `images` 传递给 Promise 的成功处理函数。

如果超时计时器触发，即超过指定的时间 `timeout`，会设定 `isTimeout` 为 `true`，并调用 `reject` 函数，传递一个超时的错误给 Promise 的失败处理函数。

最后，使用示例代码演示了如何调用 `loadImages` 函数，并处理加载完成或超时的情况。在 Promise 的 `then` 方法中，打印加载完成的图片数组；在 `catch` 方法中，打印加载错误的信息。

请注意，实际应用中，可以根据具体需求进行适当的调整和改进，如添加错误处理、优化加载方式等。以上示例仅提供了一个基本的加载图片并处理超时的框架。

下面是一个手写的示例，使用 `setTimeout` 循环多次，并提供停止循环的功能：

```javascript
function loopWithTimeout(callback, delay, maxIterations) {
  let iteration = 0;
  let timeoutId;

  function loop() {
    callback(iteration);

    if (++iteration < maxIterations) {
      timeoutId = setTimeout(loop, delay);
    }
  }

  loop();

  // 返回一个函数，用于停止循环
  return function stopLoop() {
    clearTimeout(timeoutId);
  };
}

// 示例用法
const stop = loopWithTimeout(iteration => {
  console.log('Iteration:', iteration);

  // 在第 5 次迭代后停止循环
  if (iteration === 4) {
    stop();
    console.log('Loop stopped.');
  }
}, 1000, 10);
```

在上述示例中，`loopWithTimeout` 函数接收三个参数：`callback`（每次循环的回调函数）、`delay`（延迟执行的时间间隔）和 `maxIterations`（最大循环次数）。

在函数内部，定义了 `iteration` 变量来跟踪当前的迭代次数，以及 `timeoutId` 来存储 `setTimeout` 返回的定时器 ID。

通过 `loop` 函数来执行循环操作。首先调用回调函数 `callback`，并传递当前的迭代次数 `iteration`。然后，检查是否达到了最大循环次数 `maxIterations`，如果没有，则通过 `setTimeout` 设置下一次循环的延迟，并将 `loop` 函数作为回调函数传递给 `setTimeout`。

调用 `loop` 函数来启动循环。

最后，`loopWithTimeout` 函数返回一个用于停止循环的函数。该函数会调用 `clearTimeout` 并传入保存的 `timeoutId`，从而停止后续的循环。

使用示例展示了如何调用 `loopWithTimeout` 函数，并在第 5 次迭代后停止循环。在每次迭代中，打印当前的迭代次数。在第 5 次迭代时，调用停止循环的函数，并打印一条停止循环的消息。

请注意，在实际使用中，可以根据具体需求进行适当的调整和改进，如添加错误处理、优化循环逻辑等。以上示例提供了一个基本的循环多次并提供停止功能的框架。
