写一个函数，入参是一个类数组，如果里面元素有promise（元素有可能不是promise），返回最后一个执行完的promise，如果全部reject，抛出异常  
  下面是一个函数 `getLastCompletedPromise`，它接受一个类数组作为参数，并返回最后一个执行完的 Promise。如果全部 Promise 都被拒绝（rejected），则会抛出一个异常。

```javascript
function isPromise(obj) {
  return Promise.resolve(obj) === obj;
}

async function getLastCompletedPromise(arr) {
  let lastPromise = null;

  for (let i = 0; i < arr.length; i++) {
    const item = arr[i];

    if (isPromise(item)) {
      lastPromise = item;
    } else {
      continue;
    }

    try {
      await item;
    } catch (error) {
      // 如果有一个 Promise 被拒绝，则立即抛出异常
      throw error;
    }
  }

  if (lastPromise === null) {
    throw new Error('没有找到 Promise 对象');
  }

  return lastPromise;
}
```

使用示例：

```javascript
// 示例类数组
const arr = [1, Promise.resolve(2), 3, Promise.resolve(4), Promise.reject(new Error('Rejected'))];

getLastCompletedPromise(arr)
  .then(result => {
    console.log('最后一个执行完的 Promise:', result);
  })
  .catch(error => {
    console.error('全部 Promise 被拒绝:', error);
  });
```

在上述示例中，类数组 `arr` 包含了一些数字和 Promise 对象。函数 `getLastCompletedPromise` 遍历该数组，找到最后一个 Promise 对象并等待其执行完毕。如果其中有一个 Promise 被拒绝，则立即抛出异常。在结果中，你将获得最后一个执行完的 Promise 对象。如果全部 Promise 都被拒绝，则会捕获到异常。
