```js
function jsonToTree(data, parent = null) {
	let tree = []
	for (let i = 0; i < data.length; i++) {
		if (data[i].parent_id === parent) {
			let child = data[i]
			let children = jsonToTree(data, child.id)
			if (children.length > 0) {
				child.children = children
			}
			tree.push(child)
		}
	}
	return tree
}    
```
