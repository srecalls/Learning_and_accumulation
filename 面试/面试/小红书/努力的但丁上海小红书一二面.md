```js
var flat = function (arr, n) {
	let res = []
	for (let i = 0; i < arr.length; i++) {
		if (Array.isArray(arr[i]) && n > 0) {
			res = res.concat(flat(arr[i], n - 1))
		} else {
			res.push(arr[i])
		}
	}
	return res
};
```

```js
var flat = function(arr, n) {
	return arr.reduce((total, cur) => {
		return total.concat(Array.isArray(cur) && n > 0 ? flat(cur, n - 1) : [cur])
	}, [])
}
```

```js
var isValid = function(s) {
	let map = new Map([
		[')', '('],
		['}', '{'],
		[']', '[']
	])
	let res = []
	for (let i = 0; i < s.length; i++) {
		if (map.has(s[i])) {
			if (res.length === 0 || res[res.length - 1] !== map.get(s[i])) {
				return false
			} else {
				res.pop()
			}
		} else {
			res.push(s[i])
		}
	}

}
```

```js
class MyBigInt {
	constructor(num) {
		if (typeof num === 'string') {
			this.num = num
		} else {
			throw new Error('type error')
		}		
	}
	plus(other) {
		if (other instanceof MyBigInt) {
			let temp = 0
			let res = ''
			let num1 = this.num.split('')
			let num2 = other.num.split('')
			while(num1.length || num2.length || temp) {
			temp = temp + ~~num1.pop() + ~~num2.pop()
			res = res + (temp % 10)
			temp = Math.floor(temp / 10)
		  }
		  return res
		} else {
			throw new Error('type error')
		}
	}
}
```
