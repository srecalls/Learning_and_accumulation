```javascript
function uploadImage(image) {
  return new Promise((resolve, reject) => {
    const formData = new FormData();
    formData.append('image', image);

    fetch('/upload', {
      method: 'POST',
      body: formData
    })
      .then(response => {
        // 检查响应状态码，这里假设 200 表示上传成功
        if (response.status === 200) {
          resolve();
        } else {
          reject(new Error('Upload failed'));
        }
      })
      .catch(error => {
        reject(error);
      });
  });
}

async function uploadImages(images) {
  const promises = images.map((image) => {
    return async function () {
      try {
        await uploadImage(image);
        return true; // 表示上传成功
      } catch (error) {
        return false; // 表示上传失败
      }
    }();
  });

  const results = await Promise.all(promises);

  if (results.every(result => result)) {
    // 全部上传成功，执行回调
    callback();
  } else {
    // 有上传失败的图片，进行重传
    const failedImages = images.filter((_, index) => !results[index]);
    uploadImages(failedImages);
  }
}

// 示例用法
const images = [image1, image2, image3]; // 图片数组
const callback = () => {
  console.log('All images uploaded successfully');
};

uploadImages(images);
```

在上述代码中，`uploadImage` 函数用于上传单张图片，返回一个 Promise 对象。它使用 Fetch API 将图片通过 FormData 形式发送到服务器，并根据响应状态码判断上传是否成功。

`uploadImages` 函数是主要的上传逻辑。它接受一个图片数组 `images` 作为参数，并使用 `map` 方法创建了一个由 Promise 对象组成的数组 `promises`。每个 Promise 对象表示对应图片的上传操作。

在 `uploadImages` 函数中，我们使用了 async/await 和 IIFE（立即调用函数表达式）的组合来实现图片的并发上传。同时，我们使用了 Promise.all 方法等待所有 Promise 对象的状态改变，然后根据上传结果进行后续处理。

function promiseAll(promise) {
	return new Promise((resolve, reject) => {
		if (!Array.isArray(promise)) throw new TypeError('argument must be an array')
		let resolvedCount = 0
		let resolvedResult = []
		for (let i = 0; i < promise.length; i++) {
			Promise.resolve(promise[i])
				.then(res => {
					resolvedCount++
					resolvedResult[i] = res
					if (resolvedCount === promise.length) {
						return resolve(resolvedResult)
					}
				}, err => {
					return reject(err)
				})
		}
	})
}
