## 打点
```js
// 页面曝光打点
let lastRouteTime = null; // 记录上一次页面访问的时间，用来计算页面停留时长
let lastExtra = {}; // 用来保存上一次页面曝光事件携带的参数（比如appId）
const trackPageView = (page, extra = {}) => {
  const newRouteTime = new Date().getTime();
  const fromRef = historyQueue.length > 1 ? historyQueue[historyQueue.length - 2] : '';
  const data = {
    sub_ref: lastRoutePath,
    server_time: lastRouteTime,
    duration: newRouteTime - lastRouteTime,
    refs: historyQueue.join('-'),
    from_ref: fromRef,
  };
  if (!page && lastRouteTime) {
    // page参数不传，表示用户离开了当前路由且没有命中新的路由（比如关闭页面或切换tab），立即上报最近一次路由的view事件
    // lastRouteTime = null，下次用户再次回到页面时（切换TAB）视为新的view事件
    buTrack('view', {
      ...data,
      ...lastExtra,
    });
    lastRouteTime = null;
    return;
  }
  if (
    lastRouteTime
    && page === lastRoutePath
    && JSON.stringify(extra) === JSON.stringify(lastExtra)
  ) {
    // 该页面已经记录过了，不做任何修改
    return;
  }
  if (lastRouteTime) {
    // 由于要计算duration，因此每次路由切换时，上报上一个路由的view事件
    buTrack('view', {
      ...data,
      ...lastExtra,
    });
  }
  if (page) {
    historyQueue.push(page);
    lastRoutePath = page;
    lastExtra = extra;
  }
  lastRouteTime = newRouteTime;
};
window.addEventListener('visibilitychange', () => {
  if (lastRoutePath !== 'none') {
    trackPageView();
  }
});

```
