难度：<font color ="orange">中等</font>
给你一个下标从 **0** 开始、大小为 `n x n` 的整数矩阵`grid` ，返回满足 $R_1$ 行和 $C_j$ 列相等的行列对 ($R_i$, $C_j$) 的数目。

如果行和列以相同的顺序包含相同的元素（即相等的数组），则认为二者是相等的。

 

示例 1：
![[Pasted image 20230606113109.png]]


	输入：grid = [[3,2,1],[1,7,6],[2,7,7]]
	输出：1
	解释：存在一对相等行列对：
	- (第 2 行，第 1 列)：[2,7,7]
示例 2：

![[Pasted image 20230606113126.png]]
	输入：grid = [[3,1,2,2],[1,4,4,5],[2,4,2,2],[2,4,2,2]]
	输出：3
	解释：存在三对相等行列对：
	- (第 0 行，第 0 列)：[3,1,2,2]
	- (第 2 行, 第 2 列)：[2,4,2,2]
	- (第 3 行, 第 2 列)：[2,4,2,2]
 

提示：

	n == grid.length == grid[i].length
	1 <= n <= 200
	1 <= grid[i][j] <= 10^5

```js
/**
 * @param {number[][]} grid
 * @return {number}
 */

var equalPairs = function(grid) {
    // 记录矩阵行列数
    let n = grid.length
    // 用于存储矩阵每列的数据
    let col_grid = []
    // 用于存储扁平化后的矩阵
    let flat_grid = []
    // 利用flat函数将数组扁平化
    flat_grid = grid.flat(1)
    // 用于计算矩阵每一列
    for (let i = 0; i < n; i++) {
        // 将矩阵'第一行'的对应'数'存入'列数组',方便计算
        let col = [flat_grid[i]]
        for (let j = n; j < n * n; j++) {
            // j + 1和i + 1方便数学计算,从而计算每列的数将其存入列数组
            // 比如示例1中的第1行第0列的数字'1'，下标对应在扁平化数组中对应下标3
            // 那么就(3 + 1) % 3 == (0 + 1) % 3,后面%3是为了防止6%3==3导致的不成立
            if ((j + 1) % n == ((i + 1) % n)) {
                col.push(flat_grid[j])
            }
        }
        col_grid.push(col)
    }
    let number = 0
    // 用于比较
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < n; j++) {
            if(grid[i].toString() == col_grid[j].toString()) {
                number++
            }
        }
    }
    return number
};
```
