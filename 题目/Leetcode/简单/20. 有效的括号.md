给定一个只包括` '('，')'，'{'，'}'，'['，']' `的字符串 s ，判断字符串是否有效。

有效字符串需满足：

1. 左括号必须用相同类型的右括号闭合。
2. 左括号必须以正确的顺序闭合。
3. 每个右括号都有一个对应的相同类型的左括号。
 

示例 1：

	输入：s = "()"
	输出：true
示例 2：

	输入：s = "()[]{}"
	输出：true
示例 3：

	输入：s = "(]"
	输出：false


提示：

	1 <= s.length <= 104
	s 仅由括号 '()[]{}' 组成

来源：力扣（LeetCode）
链接：https://leetcode.cn/problems/valid-parentheses

```js
/**
 * @param {string} s
 * @return {boolean}
 */

var isValid = function(s) {
    if (s.length % 2 == 1) {
        return false
    }
    const map = new Map([
        ['}', '{'],
        [')', '('],
        [']', '[',]
    ])
    let stack = []
    for (let i = 0; i < s.length; i++) {
        if (map.has(s[i])) {
            // 这里stack == 0是为了说明如果stack里没有了，但是还进来一个}这种的就直接return false
if (stack == 0 || stack[stack.length - 1] !== map.get(s[i])) {
                return false
            }
            stack.pop()
        } else {
            stack.push(s[i])
        }
    }
    return !stack.length
}
```
