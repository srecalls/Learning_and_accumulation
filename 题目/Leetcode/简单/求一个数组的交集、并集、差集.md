
## 交集
```js
const nums1 = [1, 2, 4, 5, 6, 8, 9]
const nums2 = [2, 3, 4, 5]
var intersection = function(nums1, nums2) {
	let res = nums1.filter(item => {
		return nums2.includes(item)
	})
	let final = new Set(res)
	return [...final]
}
// [2, 4, 5]
```


## 并集
### 利用交集
```js
const nums1 = [1, 2, 4, 5, 6, 8, 9]
const nums2 = [2, 3, 4, 5]
const union = function (nums1, nums2) {
	let total = nums1.concat(nums2)
	let intersection = intersection(nums1, nums2)
	let res = total.filter(item => {
		return !total.includes(intersection)
	})
	let final = new Set(res)
	return [...final]
}
// [1, 2, 4, 5, 6, 8, 9, 3]
```

### 不利用交集
```js
const nums1 = [1, 2, 4, 5, 6, 8, 9]
const nums2 = [2, 3, 4, 5]
const union = function (nums1, nums2) {
	let total = nums1.concat(nums2)
	let res = new Set(total)
	return [...res]
}
// [1, 2, 4, 5, 6, 8, 9, 3]
```

## 差集
### 利用交集
```js
const nums1 = [1, 2, 4, 5, 6, 8, 9]
const nums2 = [2, 3, 4, 5]
const differenceSet = function() {
	let section = intersection(nums1, nums2)
	const res = nums1.filter(item => {
		return !section.includes(item)
	})
	let final = new Set(res)
	return [...final]
}
[1, 6, 8, 9]
```

### 利用并集
```js
const nums1 = [1, 2, 4, 5, 6, 8, 9]
const nums2 = [2, 3, 4, 5]
const differenceSet = function() {
	let unionSet = union(nums1, nums2)
	const res = unionSet.filter(item => {
		return !nums2.includes(item)
	})
	let final = new Set(res)
	return [...final]
}
[1, 6, 8, 9]
```
