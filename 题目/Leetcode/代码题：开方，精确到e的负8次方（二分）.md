function sqrt(n) {
  if (n < 0) return NaN; // 负数没有实数根
  if (n === 0) return 0; // 特殊处理0的情况

  let left = 0; // 左边界
  let right = n; // 右边界
  let mid = 0; // 中间位置
  let lastMid = 0; // 上一次的中间位置

  while (Math.abs(mid - lastMid) > 1e-8) {
    lastMid = mid;
    mid = (left + right) / 2; // 取中点

    if (mid === 0 || mid === n) break; // 避免除数为0和无限循环的情况

    if (mid * mid > n) {
      right = mid; // 如果平方大于n，则在左半边继续查找
    } else {
      left = mid; // 如果平方小于等于n，则在右半边继续查找
    }
  }

  return mid.toFixed(8); // 保留8位小数
}
