                            zone_name: this.vmInfo.zone_info.name,
                            tags: this.vmInfo.tag.join(', '),
                            organization_type: this.vmInfo.organization_info.type || '',
                            organization_name: this.vmInfo.organization_info.name || '',
                            resource_pool_name: this.vmInfo.resource_pool_info.resource_pool_name || ''
                        }

                        this.vmInfo.disk_info.forEach(item => {
                            if (item.disk_type === 'SYSTEM_DISK') {
                                this.vmInfo.systemDiskInfo = item.disk_size + 'GB'
                                this.vmInfo.systemDiskInfo += item.category ? '(' + item.category + ')' : ''
                            } else if (item.disk_typ === 'DATA_DISK') {

                                this.vmInfo.dataDiskInfo += item.category ? '(' + item.category + ')' : ''
                            }
                        })
                        this.vmInfo = Object.assign(this.vmInfo, subInfo)
                    } else {
                        this.$bkMessage({
