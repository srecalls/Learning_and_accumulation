```js
class App extends React.Component {
	state = {
		this.count = 0
	}
	onIncrement = () => {
		console.log('事件处理程序中的this', this) // undefined
		this.setState({
			count: this.state.count + 1
		})
	}
	render() {
		return (
			<button onClick={() => this.onIncrement}>+1</button>
		)
	}
}
ReactDOM.render(<App/>, document.getElementById('root'))
```
