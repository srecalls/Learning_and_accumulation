class App extends React.Component {
	state = {
		this.count = 0
	}
	handleClick() {
		this.setState({
			count: this.state.count + 1 // 0 + 1
		})
	}
	// 钩子函数
	showComponentUpdate(nextProps, nextState) {
		// 返回false, 阻止组件重新渲染
		// return false
		console.log('最新的state: ', nextState) // 最新的状态
		console.log('this.state: ', this.state) // 更新前的状态
		// 根据条件，决定是否重新渲染组件
		return true
	}
	render() {
		return (
			<button onClick={this.handleClick}>
			+1
			</button>
		)
	}
}
ReactDOM.render(<App/>, document.getElementById('root'))

## 案例
### 随机数
![[2.避免不必要的重新渲染-2.png]]
通过state来更新
```js
class App extends React.Component {
	state = {
		number: 0
	}
	handleClick = () => {
		this.setState((state) => {
			return {
				number: Math.floor(Math.random()*3)
			}
		})
	}
	// 因为两次生成的随机数可能相同，如果相同，此时，不需要重新渲染
	showComponentUpdate(nextProps, nextState) {
		console.log('最新的state: ', nextState) // 最新的状态
		console.log('this.state: ', this.state) // 更新前的状态
		return nextState.number === this.state.number
	}
	render() {
		return (
			<div>
				<h1>随机数: {this.state.number}</h1>
				<button onClick={this.handleClick}>重新生成</button>
			</div>
		)
	}
}
```

通过props来更新
```js
class App extends React.Component {
	state = {
		number: 0
	}
	handleClick = () => {
		this.setState((state) => {
			return {
				number: Math.floor(Math.random()*3)
			}
		})
	}
	render() {
		return (
			<div>
				<button onClick={this.handleClick}>
				<Number number={this.state.number}></Number>
			</div>
		)
	}
}

class Number extends React.Component {

	// 因为两次生成的随机数可能相同，如果相同，此时，不需要重新渲染
	showComponentUpdate(nextProps, nextState) {
		console.log('最新的props: ', nextProps) // 最新的状态
		console.log('this.props: ', this.props) // 更新前的状态
		return nextProps.number === this.props.number
	}
	render() {
		return	<h1>随机数: {this.props.number}</h1>
	}
}
```
