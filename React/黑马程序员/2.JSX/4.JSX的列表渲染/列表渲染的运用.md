```jsx
const songs = [
	{id: 1, name: '痴心绝对'},
	{id: 2, name: '痴心绝对1'},
	{id: 3, name: '痴心绝对2'}
]
const list = (
	<ul>
		{songs.map(item) => <li key={item.id}>{item.name}</li>}
	</ul>
)
```

```jsx
// 列表渲染
[
	<li key={i}>Item #{i} (keep scrolling)</li>,
	<li key={i}>Item #{i} (keep scrolling)</li>,
	<li key={i}>Item #{i} (keep scrolling)</li>,
	<li key={i}>Item #{i} (keep scrolling)</li>
]
function LongSection() {
  const items = [];
  for (let i = 0; i < 50; i++) {
    items.push(<li key={i}>Item #{i} (keep scrolling)</li>);
  }
  return <ul>{items}</ul>
}
```
