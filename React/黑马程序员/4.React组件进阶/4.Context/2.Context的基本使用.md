## 例子
```js
class Parent extends React.Component {
	render() {
		return (
			<Provider value="pink">
				<div className="Parent">
					<Node/>
				</div>
			<Provider>
		)
	}
}
const Node = props => {
	return (
		<div className="Node">
			<SubNode/>
		</div>
	)
}
const SubNode = props => {
	return (
		<div className="SubNode">
			<Child/>
		</div>
	)
}
const Child = props => {
	return (
		<Consumer>
		{
			data => <span>我是子节点 -- {data}</span>
		}
		</Consumer>
	)
}
```
