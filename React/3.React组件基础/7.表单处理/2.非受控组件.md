- 说明: 借助于 ref，使用原生 DOM方式来获取表单元素值
- ref 的作用: 获取 DOM 或组件


```js
constructor() {
	super()
	// 调用React.createRef()方法创建一个ref对象
	this.txtRef = React.createRef()
}
```

```js

// 调用current方法
console.log(this.txtRef.current.value)
```



```js
class App extends React.Component {
	constructor() {
		super()
		this.txtRef = React.createRef()
		this.state = {
			txt: ''
		}
	}
	handleClick = () => {
		console.log(this.txtRef.current.value)
	}
	render() {
		return (
			<button name="txt" ref={this.txtRef} value={this.state.txt} onChange={this.handleClick}>
		)
	}
}
```
