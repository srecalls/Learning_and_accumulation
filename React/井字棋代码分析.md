import React from 'react';
import ReactDOM from 'react-dom';

// 引入useState 这个Hook
import { useState } from 'react';


const root = ReactDOM.createRoot(document.getElementById('root'));
root.render(
  <React.StrictMode>

// 创建Square组件，传入参数为value和onSquareClick
function Square({ value, onSquareClick }) {
  return (
	// 返回按钮
	// 用这里的按钮来调用Board里的事件
    <button className="square" onClick={onSquareClick}>
      {value}
    </button>
  );
}


function Board({ xIsNext, squares, onPlay }) {
  // 点击事件
  function handleClick(i) {
	// 如果已经得出胜者或者格子已经有值，则返回
    if (calculateWinner(squares) || squares[i]) {
      return;
    }
	// 这里利用squares.slice生成一个新数组
    const nextSquares = squares.slice();
    // 判断X行动还是O行动
    if (xIsNext) {
      nextSquares[i] = 'X';
    } else {
      nextSquares[i] = 'O';
    }
    // 进行handlePlay
    onPlay(nextSquares);
  }
// 通过calculatorWinner计算出true 或者false
// 从而修改显示状态
  const winner = calculateWinner(squares);
  let status;
  if (winner) {
  // 通过返回的数组里是X还O判断谁是赢者
    status = 'Winner: ' + winner;
  } else {
   // 通过返回xIsNext判断下一个是谁下
    status = 'Next player: ' + (xIsNext ? 'X' : 'O');
  }

  return (
	// 幽灵节点
    <>
      <div className="status">{status}</div>
      <div className="board-row">
        <Square value={squares[0]} onSquareClick={() => handleClick(0)} />
        <Square value={squares[1]} onSquareClick={() => handleClick(1)} />
        <Square value={squares[2]} onSquareClick={() => handleClick(2)} />
      </div>
      <div className="board-row">
        <Square value={squares[3]} onSquareClick={() => handleClick(3)} />
        <Square value={squares[4]} onSquareClick={() => handleClick(4)} />
        <Square value={squares[5]} onSquareClick={() => handleClick(5)} />
      </div>
      <div className="board-row">
        <Square value={squares[6]} onSquareClick={() => handleClick(6)} />
        <Square value={squares[7]} onSquareClick={() => handleClick(7)} />
        <Square value={squares[8]} onSquareClick={() => handleClick(8)} />
      </div>
    </>
  );
}

export default function Game() {
  // 利用useState这个Hook，设置history默认值为 数组里有一个9个空元素的数组, 这样设计是为了存储每一次下完之后当前棋盘的状态
  const [history, setHistory] = useState([Array(9).fill(null)]);
  // 利用useState这个Hook，设置currentMove为0，即当前的移动步数为0
  const [currentMove, setCurrentMove] = useState(0);
  // 通过%2来判断现在是否是X或者O的回合
  const xIsNext = currentMove % 2 === 0;
  // 设置现在的数组为history数组里的当前步数对应下标对应的棋盘数组
  const currentSquares = history[currentMove];

	// 传入
  function handlePlay(nextSquares) {
    // 设置下一步的格子为, 这里由于history为引用类型，不能直接修改其值，利用扩展运算符来进行重新创建一个
    // 这里利用slice的原因是因为，nextSquares会添加到最后，利用slice来消除后面步数的数组，这里加1是因为slice左闭右开，比如当前是0，就需要slice(0,1),然后添上下一个数组
    const nextHistory = [...history.slice(0, currentMove + 1), nextSquares];
    // 利用setHistory修改history的值为下一个history，即nextHistory，添加棋盘
    setHistory(nextHistory);
    // 利用setCurrentMove修改currentMove为数组长度-1，即最后的下标
    // 这里不用currentMove+1，是因有回退操作
    setCurrentMove(nextHistory.length - 1);
  // 这里会触发，因为组件重新渲染，导致Bored重新渲染，导致进行onPlay，从而触发console
    console.log('1', 1)
  }

	// 设置下一步跳哪的函数
  function jumpTo(nextMove) {
	// 这里会修改当前的步数, 修改后重新渲染Game，触发其中的函数
    setCurrentMove(nextMove);
  }

 // moves函数，遍历history数组，得到其中的每一次走过的棋盘和对应的部署
  const moves = history.map((squares, move) => {
    let description;
	    // 条件判断，判断其是否大于0
    if (move > 0) {
      description = 'Go to move #' + move;
      // 如果未大于0，则说明返回开始
    } else {
      description = 'Go to game start';
    }
    return (
	  // 利用map进行列表渲染要加上key
      <li key={move}>
	      // 通过jumpTo函数并带上move形参实现回到第几步
	      // description用于显示描述
        <button onClick={() => jumpTo(move)}>{description}</button>
      </li>
    );
  });

  return (
  // 大盒子
    <div className="game">
    // 棋盘盒子
      <div className="game-board">
      // 棋盘组件，传入xIsNext判断是O还是X行动，传入squares的当前棋盘，利用onPlay回调函数用于修改在Board操作后，组件内的状态（关键）
        <Board xIsNext={xIsNext} squares={currentSquares} onPlay={handlePlay} />
      </div>
      // 回退操作盒子
      <div className="game-info">
        <ol>{moves}</ol>
      </div>
    </div>
  );
}

// 计算获胜者
function calculateWinner(squares) {
  const lines = [
    [0, 1, 2],
    [3, 4, 5],
    [6, 7, 8],
    [0, 3, 6],
    [1, 4, 7],
    [2, 5, 8],
    [0, 4, 8],
    [2, 4, 6],
  ];
  // 循环数组，解构赋值获得每一项下标为a, b, c
  for (let i = 0; i < lines.length; i++) {
    const [a, b, c] = lines[i];
		// 判断所有情况，比如0,1,2一种赢法,0,4,8一种。
    if (squares[a] && squares[a] === squares[b] && squares[a] === squares[c]) {
      return squares[a];
    }
  }
  // 如果都不满足，则继续
  return null;
}
