import React from 'react';
import ReactDOM from 'react-dom/client';
import { useState } from 'react';

const root = ReactDOM.createRoot(document.getElementById('root'));
root.render(
  <React.StrictMode>
    <Game />
  </React.StrictMode>
);

function Square({value, onSquareClick}) {
	const styleObj = {
		height: 20,
		width: 20,
		backgroundColor: 'lightgreen'
	}
	return (
		<button style={ styleObj } onClick={onSquareClick}>
			{value}
		</button>
	)
}

function Board({isXTurn, square, onPlay}) {
	function handleClick(value) {
		if (getWinner(square) || square[value]) {
			return
		}
		const nextSquare = square.concat()
		if (isXTurn) {
			nextSquare[value] = 'X'
		} else {
			nextSquare[value] = 'O'
		}
		onPlay(nextSquare)
	}
	function getWinner(square) {
		const successSolution = [
			[0, 1, 2],
			[0, 3, 6],
			[0, 4, 8],
			[1, 4, 7],
			[2, 5, 8],
			[2, 4, 6],
			[3, 4, 5],
			[6, 7, 8]
		]
		for (let i = 0; i < successSolution.length; i++) {
			const [a, b, c] = successSolution[i]
			if (square[a] && square[a] === square[b] && square[a] === square[c]) {
				return true
			}
		}
		return false
	}
	let winner = getWinner(square)
	let status
	if (winner) {
		status = `the winner is ${isXTurn === 0 ? 'X' : 'O'}`
	} else {
		status = `the next Turn is ${isXTurn === 0 ? 'O' : 'X'}`
	}
	return (
		<>
			<div className='status'>{status}</div>
			<div className="firstLine">
				<Square value={square[0]} onSquareClick={() => handleClick(0)}></Square>
				<Square value={square[1]} onSquareClick={() => handleClick(1)}></Square>
				<Square value={square[2]} onSquareClick={() => handleClick(2)}></Square>
			</div>
			<div className="secondLine">
				<Square value={square[3]} onSquareClick={() => handleClick(3)}></Square>
				<Square value={square[4]} onSquareClick={() => handleClick(4)}></Square>
				<Square value={square[5]} onSquareClick={() => handleClick(5)}></Square>
			</div>
			<div className="thirdLine">
				<Square value={square[6]} onSquareClick={() => handleClick(6)}></Square>
				<Square value={square[7]} onSquareClick={() => handleClick(7)}></Square>
				<Square value={square[8]} onSquareClick={() => handleClick(8)}></Square>
			</div>
		</>

	)
}

function BackBox({history, jumpTo}) {
	function handleClick(step) {
		jumpTo(step)
	}
	const moves = history.map((item, move) => {
		return (
			<li key={move} >
				<button onClick={() => handleClick(move)}>{move}</button>
			</li>
		)
	})
	return (
		<ol>
			{moves}
		</ol>
	)
}

export default function Game() {
	const [history, setHistory] = useState([Array(9).fill(null)])
	const [currentStep, setCurrentStep] = useState(0)
	const currentSquare = history[currentStep]
	const isXTurn = currentStep % 2
	function handlePlay(nextSquare) {
		const nextHistory = [...history.slice(0, currentStep + 1), nextSquare]
		setHistory(nextHistory)
		setCurrentStep(nextHistory.length - 1)
	}
	function jumpTo(step) {
		setCurrentStep(step)
	}
	return (
		<div className="mainBox">
			<div className="chessBoard">
				<Board isXTurn={isXTurn} square={currentSquare} onPlay={handlePlay}></Board>
			</div>
			<div className="backBox">
				<BackBox history={history} jumpTo={jumpTo}></BackBox>
			</div>
		</div>
	)
}
