```js

// 导入图片
import img from './images.cat.png'

class Mouse extends React.Component {
	// 鼠标位置state
	state = {
		x: 0,
		y: 0
	}

	// 鼠标移动事件的事件处理程序
	handleMouseMove = e => {
		this.setState({
			x: e.clientX,
			y: e.clientY
		})
	}

	// 监听鼠标移动事件
	// DOM元素渲染完毕后
	componentDidMount() {
		window.addEventListener('mousemove', this.handleMouseMove)
	}

	render() {
		return this.props.render(this.state) // 这里的返回值就是下面的JSX
		// 然后通过这里render的return来进行显示
	}
}

class App extends React.Component {
	render() {
		return (
			<div>
				<h1>render props 模式</h1>
				<Mouse render={(mouse) => {
					return (
					// 返回值
						<p>
							鼠标位置: {mouse.x}, {mouse.y}
						</p>
					)
				}}
				<Mouse render={(mouse) => {
					return (
						// react脚手架中使用图片的方式
						<img src={img} alt="cat" style={{
							position: absolute,
							top: mouse.y,
							left: mouse.x
						}}>
					)
				}}
			</div>
		)
	}
}

ReactDOM.render(<App/>, document.getElementById('root'))
```
