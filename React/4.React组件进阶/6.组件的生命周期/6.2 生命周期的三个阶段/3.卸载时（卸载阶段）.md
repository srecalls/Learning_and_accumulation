class App extends React.Component {
	constructor(props) {
		super(props)
		this.state = {
			count: 0
		}
	}
	
	render() {
		return (
			<div>
			{this.state.count > 3 : (
				'cancleCount'
			) : (
				<Count count={this.state.count}></Count>
			)}
			</div>
		)
	}

}

class Counter extends React.Component {
	componentDidMount() {
		// 开启定时器
		this.timerId = setInterval(() => {
			console.log('定时器在执行')
		}, 500)
	}
	render() {
		return <h1>{this.props.count}</h1>
	}
	// 组件卸载（从页面中消失）
	componentWillUnmount(prePros) {
		console.log('子组件生命周期钩子函数: componentWillUnmount')
		// 清理定时器
		clearInterval(this.timerId)
	}
}
