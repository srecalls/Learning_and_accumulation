## 更新时（更新阶段）

```js
class App extends React.Component {
	constructor(props) {
		super(props)
		// 初始化state
		this.state = {
			count: 0
		}
	}

	handleClick = () => {
		// 2. setState() 触发 render()
		this.setState({
			count: this.state.count + 1
		})

		// 3. forceUpdate() 触发 render()
		this.forceUpdate()
	}
	// 每次组件渲染都触发
	render() {
		console.log('生命周期钩子函数: render')
		return (
			<div>
				<h1 id="title">生命周期函数</h1>
				<Counter count={this.state.count}>
				<button onClick={this.handleClick}>Button</button>
			</div>
		)
	}

}

class Counter extends React.Component {
	// 1. New Props 触发 render()
	render() {
		console.log('子组件生命周期钩子函数: render')
		return <h1>{this.props.count}</h1>
	}
	// 组件更新(完成DOM渲染)后
	componentDidUpdate(prePros) {
		console.log('子组件生命周期钩子函数: componentDidUpdate')
		// 1.发送网络请求
		// 2.DOM操作
		const title = document.getElementById('title')
		console.log(title) // <h1 id="title">生命周期函数</h1>
		
		// 注意:如果要调用 setState() 更新状态，必须要放在一个 if 条件中
		// 因为:如果直接调用 setState() 更新状态，也会导致递归更新!!!

		// 正确做法:
		// 做法:比较更新前后的props是否相同，来决定是否重新渲染组件
		if (preProps.count !=== this.props,count) {
			this.setState({})
			// 发送Ajax请求的代码
		}
		console.log('上一次的Props', preProps, '当前的Props', this.props)
		// 错误演示！！！
		this.setState()
	}
}
```
