## 创建时(挂载阶段)
![[6.2 生命周期的三个阶段-1.png]]
```js
class App extends React.Component {
	// 创建组件时，最先执行
	constructor(props) {
		super(props)
		// const title = document.getElementById('title')
		// console.log(title) null
		
		console.log('生命周期钩子函数: constructor')
		//  1.初始化state
		this.state = {
			count: 0
		}
		
		//  2.为事件处理程序绑定this
		//  this.handleCount = this.handleCount.bind(this)
	}
	
	// 每次组件渲染都会触发
	render() {
		console.log('生命周期钩子函数: render')
		// const title = document.getElementById('title')
		// console.log(title) null
		
		// 渲染UI(注意：不能调用setState())
		// 只要调用setState()就会调用render()去更新UI，如果render()里调用setState()就会造成递归更新
		return (
			<div>
				<h1 id="title">生命周期函数</h1>
			</div>
		)
	}
	// 组件挂载(完成DOM渲染)后
	componentDidMount() {
		console.log('生命周期钩子函数: componentDidMount')
		// 1.发起网络请求
		// axios.get('http://api....')
		
		// 2.DOM操作
		const title = document.getElementById('title')
		console.log(title) // <h1 id="title">生命周期函数</h1>
	}

}
```
