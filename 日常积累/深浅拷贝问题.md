// 处理热力图数据

handleData () {

	const arr = []

	for (let i = 0; i < this.mapData.length; i++) {

		arr[i] = [i % 10, Math.floor(i / 10), this.mapData[i].value]

		this.mapData[i].value = arr[i]

	}

}
