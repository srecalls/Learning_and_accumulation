如果在函数内部使用`key`作为变量名，会与函数的第二个参数`key`发生冲突，导致代码出错。

为了避免这种问题，你可以使用不同的变量名来避免命名冲突。例如，你可以将`key`参数重命名为`targetKey`，并在函数内部使用`targetKey`来代替`key`。

```js
function getParams(url,key) {
  const res = {}
  if (url.includes('?')) {
    const str = url.split('?')[1]
    console.log(str)
    const arr = str.split('&')
    console.log(arr)
    arr.forEach(item => {
      const key = item.split('=')[0]
      const val = item.split('=')[1]
      res[key] = decodeURIComponent(val) // 解码
      console.log('key', key) // user
      console.log('key', this.key) // undefined
    })
  }
  return res
}

// 测试
const user = getParams('http://www.baidu.com?user=%E9%98%BF%E9%A3%9E&age=16', '11111')
console.log(user) // { user: '阿飞', age: '16' }
```
