- 方式2:组件内 - 映射使用(推荐)
语法:
```js
import { mapState } from 'vuex'
export default {
	computed: {
		// 2. 把state里变量映射到计算属性中
		...mapState(['state里的变量名'])
	}
}
```



# SubItem映射用
```js
<template>
  <div>
      <h3>SubItem组件</h3>
      // 所以这里直接就用computed里面有的count
      <p>已知库存数: {{ count }}</p>
      <button>库存-1</button>
  </div>
</template>
<script>
// 需求1: 映射state到计算属性
// 1. 拿到辅助函数 mapState
// 2. 在computed内, ...mapState(['state变量名'])
// 3. 当计算属性使用
import { mapState } from 'vuex'
// let r = mapState(['count']) // 提取store里的state叫count的变量
// r 是一个对象 {count : 内置函数}
// 打断点看函数里return什么值,发现state['count']
// console.log(r); // 返回值: {count: 函数体(return state里count的值)}
export default {
  // 3. 把映射过来的对象, 展开放入到computed里(结构是对应的)
  // 都是变量() { return 值 }
  computed: {
    // 映射count, 得到对象展开, 合并到计算属性中
    // 要哪个变量,就映射过来哪一个,不同的用逗号隔开
    ...mapState(['count'])
    // 运行后相当于下面样子
    // count: function() {
	//     return this.$store.state.count
	// }
    // 对应上面r是一个对象,返回值: {count: 函数体(return state里count的值)},
    // 所以展开就是 count : { return } 跟computed一样
	...mapState(list:'goodsList') // 自定义属性名list, 映射vuex里的goodsList变量值
  },
}
</script>
```
