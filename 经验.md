编辑器json折叠；

```js
jsonFoldingRangeProvider() {
    this.monaco.languages.registerFoldingRangeProvider('json', {
        provideFoldingRanges: function(model, context, token) {
            const foldingRanges = []
            const stack = []
            for (let lineNumber = 1; lineNumber <= model.getLineCount(); lineNumber++) {
                const lineText = model.getLineContent(lineNumber)
                if (lineText.trim().startsWith('{') || lineText.trim().startsWith('[') || lineText.endsWith('{') || lineText.endsWith('[')) {
                    stack.push(lineNumber)
                } else if (lineText.trim().startsWith('}') || lineText.trim().startsWith(']')) {
                    const startLineNumber = stack.pop()
                    foldingRanges.push({
                        start: startLineNumber,
                        end: lineNumber - 1
                    })
                }
            }
            return foldingRanges
        }
    })
}
```
