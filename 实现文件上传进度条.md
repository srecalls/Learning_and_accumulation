import { get, post, upload } from 'common/fetch';
import { message } from 'antd';
import axios from 'axios';

export const uploadImg = ({
  data,
  file,
  headers,
  onError,
  onProgress,
  onSuccess,
  withCredentials,
}) => {
  const formData = new FormData();
  if (data) {
    Object.keys(data).forEach((key) => {
      formData.append(key, data[key]);
    });
  }
  formData.append('image', file);

  axios
    .post('/doc-api/upload/image', formData, {
      withCredentials,
      headers,
      onUploadProgress: ({
        total,
        loaded,
      }) => {
        onProgress({
          percent: Number(Math.round((loaded / total) * 100).toFixed(2)),
        }, file);
      },
    })
    .then(({
      data: response,
    }) => {
      console.log('response ------>', response, file);
      if (response.code === 0) {
        const ret = {
          url: response.data,
          screenType: data.type === 15 ? 1 : 0,
        };
        onSuccess(ret, file);
      } else {
        onError(response.msg);
        message.error(response.msg);
      }
    })
    .catch((err) => {
      onError(err);
      message.error(err);
    });

  return {
    abort() { },
  };
};

export const getVideoUrl = async (params) => {
  const response = await upload('/upload/getVideoUrl', params);

  return response.data;
};

export const saveData = (params) => post('/app/promoteresource/add', params)
  .then((r) => {
    if (r.code === 0) return Promise.resolve(r.data);
    return Promise.reject(new Error(`server side error with code ${r.code}`));
  })
  .catch((e) => Promise.reject(e));

#! index.jsx
import React, { useState, useRef, useEffect } from 'react'
import {
  Input,
  Form,
  Upload,
  Icon,
  Button,
  Progress,
  message,
} from 'antd'
import Breadcrumb from 'components/breadcrumb'
import { uploadImg, getVideoUrl, saveData } from 'apis/resources'
import { IMAGE_TYPE, resumableUploader, removeVideoItem, deletePoster, getVideoItem, removePicList } from './helper'
import { removeFileItem } from 'antd/es/upload/utils'
import CloseIcon from './components/closeIcon'
import classnames from 'classnames';
import style from './index.module.less';

const ResourcesPage = () => {
  const [packageName, setPackageName] = useState('')
  const [horizontalPicList, setHorizontalPicList] = useState([])
  const [verticalPicList, setVerticalPicList] = useState([])
  const [horizontalVideoList, setHorizontalVideoList] = useState([])
  const [verticalVideoList, setVerticalVideoList] = useState([])
  const [isSaving, setIsSaving] = useState(false);
  const horizontalVideoEl = useRef()
  const verticalVideoEl = useRef()
  const hVideoList = useRef([])
  const vVideoList = useRef([])

  useEffect(() => {
    resumableUploader({
      container: horizontalVideoEl.current,
      fileType: 3,
      onBeforeUpload: (file) => {
        if (getVideoItem(file.uniqueIdentifier, hVideoList.current)) {
          message.error('请勿上传重复视频')
          return -1
        }
        if (file.type !== 'video/mp4') {
          message.error('上传的素材不是视频')
          return -1
        }
        const videoInfo = {
          uniqueIdentifier: file.uniqueIdentifier,
          progress: 0
        }
        hVideoList.current = [...hVideoList.current, videoInfo]
        setHorizontalVideoList([...hVideoList.current])
      },
      onSuccess: async (resumableIdentifier, file) => {
        const data = await getVideoUrl({ resumableIdentifier })
        if (data.screenType === 1) {
          const removedList = removeFileItem(file, hVideoList.current)
          hVideoList.current = removedList
          message.error('请上传正确的视频类型')
          setHorizontalVideoList([...hVideoList.current])
          return
        }
        const videoInfo = getVideoItem(file.file.uniqueIdentifier, hVideoList.current)
        if(!videoInfo) {
          file.abort()
          return
        }
        videoInfo.videoUrl = data.url
        videoInfo.screenType = data.screenType
        // videoInfo.pictureUrl = ''
        videoInfo.progress = 1
        setHorizontalVideoList([...hVideoList.current])
      },
      onProgress: (progress, file) => {
        if (file.file.type !== 'video/mp4') return
        const videoInfo = getVideoItem(file.file.uniqueIdentifier, hVideoList.current)
        if(!videoInfo) {
          file.abort()
          return
        }
        if (videoInfo.progress === 1) return
        videoInfo.progress = progress
        setHorizontalVideoList([...hVideoList.current])
      }
    })

    resumableUploader({
      container: verticalVideoEl.current,
      fileType: 3,
      onBeforeUpload: (file) => {
        if (getVideoItem(file.uniqueIdentifier, vVideoList.current)) {
          message.error('请勿上传重复视频')
          return -1
        }
        if (file.type !== 'video/mp4') {
          message.error('上传的素材不是视频')
          return -1
        }
        const videoInfo = {
          uniqueIdentifier: file.uniqueIdentifier,
          progress: 0
        }
        vVideoList.current = [...vVideoList.current, videoInfo]
        setVerticalVideoList([...vVideoList.current])
      },
      onSuccess: async (resumableIdentifier, file) => {
        const data = await getVideoUrl({ resumableIdentifier })
        if (data.screenType === 0) {
          const removedList = removeFileItem(file, vVideoList.current)
          vVideoList.current = removedList
          message.error('请上传正确的视频类型')
          setVerticalVideoList([...vVideoList.current])
          return
        }
        const videoInfo = getVideoItem(file.file.uniqueIdentifier, vVideoList.current)
        if(!videoInfo) {
          file.abort()
          return
        }
        videoInfo.videoUrl = data.url
        videoInfo.screenType = data.screenType
        // videoInfo.pictureUrl = ''
        videoInfo.progress = 1
        setVerticalVideoList([...vVideoList.current])
      },
      onProgress: (progress, file) => {
        if (file.file.type !== 'video/mp4') return
        const videoInfo = getVideoItem(file.file.uniqueIdentifier, vVideoList.current)
        if(!videoInfo) {
          file.abort()
          return
        }
        if (videoInfo.progress === 1) return
        videoInfo.progress = progress
        setVerticalVideoList([...vVideoList.current])
      },
    })
  }, [])

  const handleChange = (type, videoInfo, { file, fileList }) => {
    switch (type) {
      case IMAGE_TYPE.SCREENSHOT_HORIZATAL:
        if (file.status === 'error') {
          const removedList = removeFileItem(file, fileList)
          setHorizontalPicList([...removedList])
        } else {
          if (fileList.filter(pic => pic.name === file.name).length > 1) {
            message.error('请勿上传重复图片')
          } else {
            setHorizontalPicList([...fileList])
          }
        }
        break
      case IMAGE_TYPE.SCREENSHOT_VERTICAL:
        if (file.status === 'error') {
          const removedList = removeFileItem(file, fileList)
          setVerticalPicList([...removedList])
        } else {
          if (fileList.filter(pic => pic.name === file.name).length > 1) {
            message.error('请勿上传重复图片')
          } else {
            setVerticalPicList([...fileList])
          }
          setVerticalPicList([...fileList])
        }
        break
      case IMAGE_TYPE.VIDEO_POSTER_HORIZATAL:
        if (file.status === 'done') {
          videoInfo.pictureUrl = file.response?.url
          setHorizontalVideoList([...horizontalVideoList])
        }
        break
      case IMAGE_TYPE.VIDEO_POSTER_VERTICAL:
        if (file.status === 'done') {
          videoInfo.pictureUrl = file.response?.url
          setVerticalVideoList([...verticalVideoList])
        }
        break
      default:
        break;
    }
  }

  const handleClear = () => {
    setPackageName()
    setHorizontalPicList([])
    setVerticalPicList([])
    setHorizontalVideoList([])
    setVerticalVideoList([])
    hVideoList.current = []
    vVideoList.current = []
  }

  const handleSave = () => {
    const picList = horizontalPicList
      .map(
        pic => pic.response
      )
      .concat(
        verticalPicList
          .map(pic => pic.response)
      ).map(
        pic => {
          return {
            ...pic,
            url: pic.url.match(/(AppStore.*)/g)[0]
          }
        }
      )
    const data = {
      packageName,
      videoResource: JSON.stringify(
        horizontalVideoList
          .concat(verticalVideoList)
          .map(
            video => {
              return {
                screenType: video.screenType,
                videoUrl: video?.videoUrl?.match(/(AppStore.*)/g)?.[0],
                pictureUrl: video?.pictureUrl?.match(/(AppStore.*)/g)?.[0]
              }
            }
          )
      ),
      pictureResource: JSON.stringify(picList)
    }

    setIsSaving(true);
    saveData(data)
      .then(data => {
        message.success('保存成功');
        handleClear();
      })
      .finally(() => {
        setIsSaving(false);
      })
  }

  const handleDeleteVideo = (video, videoList, setMethod, ref) => {
    const removedVideoList = removeVideoItem(video, videoList)
    ref.current = removedVideoList
    setMethod([...removedVideoList])
  }

  const handleDeletePoster = (video, videoList, setMethod, ref) => {
    const removedPosterList = deletePoster(video, videoList)
    ref.current = removedPosterList
    setMethod([...removedPosterList])
  }

  const handleDeletePic = (pic, picList, setMethod) => {
    const removedPicList = removePicList(pic, picList)
    setMethod([...removedPicList])
  }


  return (
    <div className={classnames(['base-container', style.resources_upload_manager])}>
      <Breadcrumb data={['素材库管理', '上传管理']} />
      <Form>
        <Form.Item label="包名">
          <Input value={packageName} onChange={e => { setPackageName(e.target.value) }} placeholder="请输入包名" />
        </Form.Item>
        <Form.Item label="视频">
          <div
            className="ant-upload ant-upload-select-picture-card"
            ref={horizontalVideoEl}
            style={{
              display: 'flex',
              flexDirection: 'column',
              justifyContent: 'center'
            }}
          >
            <Icon type="plus" />
            <div className="ant-upload-text">上传横屏视频</div>
          </div>
          <div
            className="ant-upload ant-upload-select-picture-card"
            ref={verticalVideoEl}
            style={{
              display: 'flex',
              flexDirection: 'column',
              justifyContent: 'center'
            }}
          >
            <Icon type="plus" />
            <div className="ant-upload-text">上传竖屏视频</div>
          </div>
          <div style={{ whiteSpace: 'nowrap' }}>不大于5min, 建议尺寸1280X720, 1920X1080</div>
        </Form.Item>
        <Form.Item label="横屏视频">
          {
            horizontalVideoList.map(videoInfo => <div style={{ display: 'inline-block' }} key={videoInfo.uniqueIdentifier}>
              <div
                style={{
                  width: '104px',
                  height: '104px',
                  display: 'flex',
                  justifyContent: 'center',
                  alignItems: 'center',
                  marginRight: '8px',
                  marginBottom: '8px',
                  position: 'relative',
                  border: '1px rgb(217, 217, 217) dashed',
                  borderRadius: '4px',
                  background: 'rgb(250, 250, 250)',
                  padding: '8px'
                }}>
                {
                  videoInfo.progress === 1 && videoInfo.videoUrl
                    ?
                    <>
                      <video style={{ maxWidth: '100%', maxHeight: '100%' }} src={videoInfo.videoUrl} onError={ e => { e.persist(); e.target.load()} } />
                      <CloseIcon onClick={handleDeleteVideo.bind(null, videoInfo, horizontalVideoList, setHorizontalVideoList, hVideoList)} />
                    </>
                    :
                    <>
                      <Progress percent={Number((videoInfo.progress * 100).toFixed(1))} />
                      <CloseIcon onClick={handleDeleteVideo.bind(null, videoInfo, horizontalVideoList, setHorizontalVideoList, hVideoList)} />
                    </>
                }
              </div>
              <Upload
                listType="picture-card"
                customRequest={uploadImg}
                showUploadList={false}
                disabled={!!videoInfo.pictureUrl}
                data={{ type: IMAGE_TYPE.VIDEO_POSTER_HORIZATAL }}
                onChange={handleChange.bind(null, IMAGE_TYPE.VIDEO_POSTER_HORIZATAL, videoInfo)}
              >
                {
                  videoInfo.pictureUrl
                    ? <div style={{
                      width: '88px',
                      height: '88px',
                      position: 'relative',
                      display: 'flex',
                      justifyContent: 'center',
                      alignItems: 'center'
                    }}>
                      <img style={{ maxWidth: '100%', maxHeight: '100%' }} src={videoInfo.pictureUrl} />
                      <CloseIcon onClick={handleDeletePoster.bind(null, videoInfo, horizontalVideoList, setHorizontalVideoList, hVideoList)} />
                    </div>
                    : <div>
                      <Icon type="plus" />
                      <div className="ant-upload-text">上传视频封面</div>
                    </div>
                }
              </Upload>
            </div>)
          }
        </Form.Item>
        <Form.Item label="竖屏视频">
          {
            verticalVideoList.map(videoInfo => <div style={{ display: 'inline-block' }} key={videoInfo.uniqueIdentifier}>
              <div
                style={{
                  width: '104px',
                  height: '104px',
                  display: 'flex',
                  justifyContent: 'center',
                  alignItems: 'center',
                  marginRight: '8px',
                  marginBottom: '8px',
                  position: 'relative',
                  border: '1px rgb(217, 217, 217) dashed',
                  borderRadius: '4px',
                  background: 'rgb(250, 250, 250)',
                  padding: '8px'
                }}
              >
                {
                  videoInfo.progress === 1 && videoInfo.videoUrl
                    ?
                    <>
                      <video style={{ maxWidth: '100%', maxHeight: '100%' }} src={videoInfo.videoUrl} preload="auto" onError={ e => { e.persist(); e.target.load()} } />
                      <CloseIcon onClick={handleDeleteVideo.bind(null, videoInfo, verticalVideoList, setVerticalVideoList, vVideoList)} />
                    </>
                    :
                    <>
                      <Progress percent={Number((videoInfo.progress * 100).toFixed(1))} />
                      <CloseIcon onClick={handleDeleteVideo.bind(null, videoInfo, verticalVideoList, setVerticalVideoList, vVideoList)} />                    
                    </>

                onChange={handleChange.bind(null, IMAGE_TYPE.VIDEO_POSTER_VERTICAL, videoInfo)}

                {
                  videoInfo.pictureUrl

                      width: '88px',
                      height: '88px',
                      position: 'relative',

                      justifyContent: 'center',
                      alignItems: 'center'

                      <img style={{ maxWidth: '100%', maxHeight: '100%' }} src={videoInfo.pictureUrl} />
                      <CloseIcon onClick={handleDeletePoster.bind(null, videoInfo, verticalVideoList, setVerticalVideoList, vVideoList)} />
